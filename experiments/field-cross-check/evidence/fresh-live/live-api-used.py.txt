def save_stats():
    (OUT/'provider-budget.json').write_text(json.dumps(statistics, indent=2)+'\n')

atexit.register(save_stats)

def guard(request):
    if request.url.host != 'api.deepseek.com':
        raise BackendError('evaluation provider host refused')
    with lock:
        if statistics['provider_request_attempts'] >= 30:
            raise BackendError('evaluation provider request cap reached')
        statistics['provider_request_attempts'] += 1
        save_stats()

def response_metadata(response):
    response.read()
    try:
        payload = response.json()
    except ValueError:
        return
    with lock:
        returned_model = payload.get('model')
        if isinstance(returned_model, str) and key not in returned_model:
            statistics.setdefault('returned_model_identifiers', [])
            if returned_model not in statistics['returned_model_identifiers']:
                statistics['returned_model_identifiers'].append(returned_model)
        usage = payload.get('usage') or {}
        for name in ('prompt_tokens', 'completion_tokens', 'total_tokens'):
            value = usage.get(name)
            if isinstance(value, int):
                statistics.setdefault('usage_tokens', {}).setdefault(name, 0)
                statistics['usage_tokens'][name] += value
        save_stats()

shared_client = openai.OpenAI(api_key=key, base_url='https://api.deepseek.com', max_retries=0,

original=Compiler._checked

def checked(self, docs, cp, offer, doc_hash):
    rule_result=RulesBackend().extract_offer(docs)
    rules=build_offer(docs,cp,rule_result.extraction,RULES_MODEL_ID,PROMPT_VERSION)
    old=baseline.check_against_rules(offer,rules)
    new=original(self,docs,cp,offer,doc_hash)
    payload=json.dumps({'doc_hash':doc_hash,'documents':[d.model_dump(mode='json') for d in docs],
                        'model':offer.model_dump(mode='json'),'rules':rules.model_dump(mode='json'),
                        'before':old.model_dump(mode='json'),'after':new.model_dump(mode='json')},ensure_ascii=False)
    if key in payload:
        raise BackendError('refused to record a credential')
    with lock:
        with (OUT/'reader-shadow.jsonl').open('a') as f:
            f.write(payload+'\n')
    return new
